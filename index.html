function calcularDiagnosticoLegal(c) {

  const anyG = Number(c.ano_grabacion) || 0;

  const anyM = Number(c.ano_muerte_autor) || 0;

  const out = {};

  const ovE = c.override_europa || '';

  const ovM = c.override_mundial || '';

  const ovS = c.override_sgae || '';


  if (anyG > 0 && anyG <= (ANNO_ACTUAL - 70)) {

    out.estat_master_europa = { ok: true, text: '🟢 DOMINI PÚBLIC A EUROPA (Cost 0€)', base: 'Gravació de fa més de 70 anys (≤ ' + (ANNO_ACTUAL-70) + ')' };

  } else {

    out.estat_master_europa = { ok: false, text: '🔴 MÀSTER COMERCIAL / PROTEGIT', base: anyG ? 'Gravació posterior a ' + (ANNO_ACTUAL-70) : 'Any de gravació no informat' };

  }

  if (anyG > 0 && anyG < 1923) {

    out.estat_master_mundial = { ok: true, text: '🟢 DOMINI PÚBLIC MUNDIAL (Cost 0€)', base: 'Anterior a 1923 (Music Modernization Act)' };

  } else {

    out.estat_master_mundial = { ok: false, text: '🔴 REQUEREIX CONTRACTE COMERCIAL MUNDIAL', base: anyG ? 'Posterior a 1922' : 'Any de gravació no informat' };

  }

  if (anyM > 0 && anyM <= (ANNO_ACTUAL - 80)) {

    out.estat_composicio_sgae = { ok: true, text: '🟢 COMPOSICIÓ EN DOMINI PÚBLIC', base: 'Autor mort fa més de 80 anys (≤ ' + (ANNO_ACTUAL-80) + ')' };

  } else {

    out.estat_composicio_sgae = { ok: false, text: '🔴 COMPOSICIÓ PROTEGIDA (SGAE)', base: anyM ? 'Autor mort després de ' + (ANNO_ACTUAL-80) : 'Any de defunció no informat' };

  }

  if (ovE === 'ok') out.estat_master_europa = { ok: true, text: '🟢 DOMINI PÚBLIC A EUROPA (manual)', base: c.override_nota || 'Override' };

  if (ovE === 'ko') out.estat_master_europa = { ok: false, text: '🔴 MÀSTER PROTEGIT (manual)', base: c.override_nota || 'Override' };

  if (ovM === 'ok') out.estat_master_mundial = { ok: true, text: '🟢 DOMINI PÚBLIC MUNDIAL (manual)', base: c.override_nota || 'Override' };

  if (ovS === 'ko') out.estat_composicio_sgae = { ok: false, text: '🔴 COMPOSICIÓ PROTEGIDA (manual)', base: c.override_nota || 'Override' };


  const masterOk = out.estat_master_mundial.ok || out.estat_master_europa.ok;

  const sgaeOk = out.estat_composicio_sgae.ok;

  if (out.estat_master_mundial.ok && sgaeOk) out.accio_requerida = '✅ APTE: Integrar directament al muntatge.';

  else if (out.estat_master_europa.ok && !sgaeOk) out.accio_requerida = '⚠️ ATENCIÓ: Àudio lliure a Europa, però cal pagar SGAE.';

  else if (sgaeOk && !masterOk) out.accio_requerida = '⚠️ ATENCIÓ: Partitura lliure, però cal negociar el màster.';

  else out.accio_requerida = '❌ NEGOCIACIÓ COMERCIAL: Cal pressupost de Màster i Editorial.';

  if (out.estat_master_mundial.ok && sgaeOk) out.color = 'ok';

  else if (!out.estat_master_europa.ok && !sgaeOk) out.color = 'ko';

  else out.color = 'warn';

  return out;

}

function classificaCobertura(cobertura, override) {

  const ov = (override || '').trim();

  if (ov === 'ok') return { estat: 'ok', text: '🟢 DOMINI PÚBLIC (override)', short: 'LLIURE' };

  if (ov === 'warn') return { estat: 'warn', text: '🟡 CC (override)', short: 'CC' };

  if (ov === 'ko') return { estat: 'ko', text: '🔴 PROTEGIT (override)', short: 'PROTEGIT' };

  if (ov === 'unk') return { estat: 'unk', text: '⚪ DESCONEGUT (override)', short: 'DESCONEGUT' };


  const c = (cobertura || '').toLowerCase().trim();

  if (!c) return { estat: 'unk', text: '⚪ DESCONEGUT', short: 'DESCONEGUT' };


  // Domini públic explícit

  if (/publicdomain\/(mark|zero|certification)/i.test(cobertura) ||

      /\b(pdm 1\.0|cc0 1\.0|cc-0|cc 0|cczero)\b/i.test(c) ||

      /\b(domini públic|public domain|dominio público|gemeinfrei|domaine public)\b/i.test(c) ||

      /prelinger/i.test(c) ||

      /no copyright|sense drets|sin derechos|no rights reserved/i.test(c)) {

    return { estat: 'ok', text: '🟢 DOMINI PÚBLIC / LLIURE', short: 'LLIURE' };

  }

  // Creative Commons amb atribució

  if (/\b(cc[-\s]?by|creative commons|atribució|attribution)\b/i.test(c)) {

    return { estat: 'warn', text: '🟡 CREATIVE COMMONS (cal atribució)', short: 'CC' };

  }

  // Protegit

  if (/\b(copyright|©|tots els drets|all rights reserved|protegit|amb drets|sota drets)\b/i.test(c)) {

    return { estat: 'ko', text: '🔴 PROTEGIT (requereix llicència)', short: 'PROTEGIT' };

  }

  return { estat: 'unk', text: '⚪ ' + cobertura.slice(0, 60), short: 'ALTRES' };

}

// --- Importació CSV/TSV ---

function parsejaCSV(text) {

  const firstLine = text.split(/\r?\n/)[0] || '';

  let sep = ',';

  if (firstLine.includes('\t')) sep = '\t';

  else if (firstLine.split(';').length > firstLine.split(',').length) sep = ';';

  const linies = text.split(/\r?\n/).filter(l => l.trim() !== '');

  return linies.map(l => {

    const camps = []; let actual = ''; let dins = false;

    for (let i = 0; i < l.length; i++) {

      const ch = l[i];

      if (ch === '"' && !dins) { dins = true; continue; }

      if (ch === '"' && dins) { if (l[i+1] === '"') { actual += '"'; i++; } else dins = false; continue; }

      if (ch === sep && !dins) { camps.push(actual); actual = ''; continue; }

      actual += ch;

    }

    camps.push(actual);

    return camps.map(c => c.trim().replace(/^"|"$/g, ''));

  });

}

function normalitzaCapcalera(h) {

  const n = h.toLowerCase().trim().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/[^a-z0-9]+/g, '_');

  if (/titol|titulo|title/.test(n)) return 'titol';

  if (/link|url|font|proced|origen|source|enlace/.test(n)) return 'link_procedencia';

  if (/cobert|derech|right|copyright|llicencia|licencia|license/.test(n)) return 'cobertura';

  if (/any|ano|year|data|date|fecha/.test(n)) return 'any_data';

  if (/tipus|tipo|type/.test(n)) return 'tipus';

  if (/autor|author|fotograf|photograph/.test(n)) return 'autor';

  if (/notes|notas|observa/.test(n)) return 'notes';

  if (/etiquet|tag/.test(n)) return 'etiquetes';

  if (/estat|estado|status/.test(n)) return 'estat_intern';

  return null;

}


async function importaImatges(text) {

  if (!text || !text.trim()) { alert('Enganxa primer les files.'); return; }

  const files = parsejaCSV(text);

  if (files.length < 2) { alert('Cal capçalera + almenys una fila.'); return; }

  const capcalera = files[0].map(normalitzaCapcalera);

  if (!capcalera.includes('titol')) { alert('No s\'ha detectat la columna "títol". Revisa la capçalera.'); return; }

  const dades = [];

  for (let i = 1; i < files.length; i++) {

    const obj = { estat_intern: 'pendent', created_by: me, updated_by: me, updated_at: new Date().toISOString() };

    files[i].forEach((v, j) => { const k = capcalera[j]; if (k) obj[k] = v; });

    if (obj.titol && obj.titol.trim()) dades.push(obj);

  }

  if (!dades.length) { alert('Cap fila vàlida.'); return; }

  if (!confirm('Importar ' + dades.length + ' imatges? Aquesta acció no es pot desfer.')) return;

  const r = await db.from('image_rights').insert(dades).select();

  if (r.error) { alert('Error important: ' + r.error.message); return; }

  imageRights = (r.data || []).concat(imageRights);

  alert('✅ ' + r.data.length + ' imatges importades.');

  render();

}

async function desaCredit(cr) {

  const payload = {

    ordre: cr.ordre != null ? cr.ordre : credits.length,

    tipus: cr.tipus || 'imatge',

    arxiu: cr.arxiu || '', atribucio: cr.atribucio || '',

    titol: cr.titol || '', any_text: cr.any_text || '',

    frase: cr.frase || '', nota: cr.nota || '',

    updated_at: new Date().toISOString(), updated_by: me

  };

  if (cr.id) {

    const r = await db.from('credits').update(payload).eq('id', cr.id).select().single();

    if (r.error) { alert('Error: ' + r.error.message); return; }

    const i = credits.findIndex(x => x.id === cr.id);

    if (i >= 0) credits[i] = r.data;

    currentCredit = null; editingCredit = false; render();

  } else {

    payload.created_by = me;

    const r = await db.from('credits').insert(payload).select().single();

    if (r.error) { alert('Error: ' + r.error.message); return; }

    credits.push(r.data);

    currentCredit = null; editingCredit = false; render();

  }

}


async function autoGeneraCredits() {

  const existents = new Set(credits.map(c => (c.arxiu||'').toLowerCase() + '|' + (c.atribucio||'').toLowerCase()));

  const nous = [];


  // Des de música: agafem les que tinguin autor o font (link)

  musicRights.forEach(m => {

    const arxiu = extrauArxiu(m.link_descarga) || 'Arxiu musical';

    const atrib = m.autor || 'Anònim';

    const key = arxiu.toLowerCase() + '|' + atrib.toLowerCase();

    if (existents.has(key)) return;

    existents.add(key);

    nous.push({

      ordre: credits.length + nous.length,

      tipus: 'musica',

      arxiu,

      atribucio: atrib,

      titol: m.titulo || '',

      any_text: m.ano_grabacion ? String(m.ano_grabacion) : '',

      frase: '', nota: '',

      created_by: me, updated_by: me

    });

  });


  // Des d'imatges: només les aprovades

  imageRights.filter(i => (i.estat_intern || '') === 'aprovada').forEach(img => {

    const arxiu = extrauArxiu(img.link_procedencia) || 'Arxiu d\'imatge';

    const atrib = img.autor || 'Anònim';

    const key = arxiu.toLowerCase() + '|' + atrib.toLowerCase();

    if (existents.has(key)) return;

    existents.add(key);

    nous.push({

      ordre: credits.length + nous.length,

      tipus: 'imatge',

      arxiu,

      atribucio: atrib,

      titol: img.titol || '',

      any_text: img.any_data || '',

      frase: '', nota: '',

      created_by: me, updated_by: me

    });

  });


  if (!nous.length) { alert('No hi ha res nou per afegir. O bé les imatges no estan marcades com "Aprovada".'); return; }

  if (!confirm('Afegir ' + nous.length + ' crèdits nous a partir de música i imatges aprovades?')) return;

  const r = await db.from('credits').insert(nous).select();

  if (r.error) { alert('Error: ' + r.error.message); return; }

  credits = credits.concat(r.data);

  alert('✅ ' + r.data.length + ' crèdits afegits.');

  render();

}


function extrauArxiu(url) {

  if (!url) return '';

  try {

    const u = new URL(url);

    const h = u.hostname.replace(/^www\./, '');

    const map = {

      'commons.wikimedia.org': 'Wikimedia Commons',

      'archive.org': 'Internet Archive',

      'bne.es': 'Biblioteca Nacional de España',

      'bdh.bne.es': 'Biblioteca Digital Hispánica (BNE)',

      'europeana.eu': 'Europeana',

      'loc.gov': 'Library of Congress',

      'flickr.com': 'Flickr Commons',

      'youtube.com': 'YouTube',

      'vimeo.com': 'Vimeo'

    };

    for (const k in map) if (h.includes(k)) return map[k];

    return h;

  } catch(e) { return ''; }

}


function textCreditsFinals() {

  return creditFrase + '\n\n' + linies.join('\n');

}

}

function exportaSubtitols(d, format) {

  const tmp = document.createElement('div'); tmp.innerHTML = d.content || '';

  const tsN = Array.from(tmp.querySelectorAll('.ts'));

  if (!tsN.length) { alert('Sense marques ⏱️.'); return; }

  const items = tsN.map(n => ({ t: parseFloat(n.dataset.t) || 0, node: n }));

  const cues = [];

  for (let i = 0; i < items.length; i++) {

    const s = items[i]; const e = items[i+1] ? items[i+1].t : s.t + 3;

    let txt = ''; let cur = s.node.nextSibling; const stop = items[i+1] ? items[i+1].node : null;

    while (cur && cur !== stop) { txt += (cur.textContent || ''); cur = cur.nextSibling; }

    txt = txt.replace(/\s+/g, ' ').trim();

    if (txt) cues.push({ start: s.t, end: Math.max(e, s.t + 0.5), text: txt });

  }

  if (!cues.length) { alert('Sense text entre marques.'); return; }

  const srt = t => { const h=Math.floor(t/3600),m=Math.floor((t%3600)/60),s=Math.floor(t%60),ms=Math.round((t-Math.floor(t))*1000); return String(h).padStart(2,'0')+':'+String(m).padStart(2,'0')+':'+String(s).padStart(2,'0')+','+String(ms).padStart(3,'0'); };

  let out = format === 'vtt' ? 'WEBVTT\n\n' : '';

  cues.forEach((c, i) => { const tm = (format === 'vtt') ? srt(c.start).replace(',', '.') + ' --> ' + srt(c.end).replace(',', '.') : srt(c.start) + ' --> ' + srt(c.end); out += (i+1) + '\n' + tm + '\n' + c.text + '\n\n'; });

  descarregaTxt(out, (d.title || 'subtitols').replace(/[^a-z0-9._-]+/gi,'_') + '.' + format);

}

function dataRelativa(isoStr) {

  if (!isoStr) return '—';

  const d = new Date(isoStr); const n = new Date();

  const min = Math.floor((n-d)/60000), h = Math.floor(min/60), dies = Math.floor(h/24);

  if (min < 1) return 'ara mateix';

  if (min < 60) return 'fa ' + min + ' min';

  if (h < 24) return 'fa ' + h + ' h';

  if (dies === 1) return 'ahir';

  if (dies < 7) return 'fa ' + dies + ' dies';

  return d.toLocaleDateString('ca-ES');

}

function fileKind(mime, name) {

  if (!mime && name) mime = name.split('.').pop().toLowerCase();

  if (typeof mime === 'string') {

    if (mime.startsWith('image/')) return 'image';

    if (mime.startsWith('audio/')) return 'audio';

    if (mime.startsWith('video/')) return 'video';

    if (mime === 'application/pdf' || mime === 'pdf') return 'pdf';

    if (mime.includes('word') || mime === 'doc' || mime === 'docx') return 'doc';

  }

  return 'file';

}
